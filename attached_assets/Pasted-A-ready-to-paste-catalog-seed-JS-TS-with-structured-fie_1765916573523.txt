2) Auto-extract images (and optionally title/description) in Replit

Create scrapeProductImages.mjs:

// scrapeProductImages.mjs
// Run: node scrapeProductImages.mjs
import fs from "node:fs";

const INPUT_FILE = "./productCatalog.seed.json";   // export your products array to JSON
const OUTPUT_FILE = "./productCatalog.withImages.json";

async function fetchText(url) {
  const res = await fetch(url, {
    redirect: "follow",
    headers: {
      "user-agent":
        "Mozilla/5.0 (compatible; CatalogBot/1.0; +https://example.com/bot)",
      "accept": "text/html,application/xhtml+xml",
    },
  });
  if (!res.ok) throw new Error(`HTTP ${res.status} for ${url}`);
  return await res.text();
}

function pickMeta(html, propertyOrName) {
  // property="og:image" OR name="description"
  const propRe = new RegExp(
    `<meta[^>]+(?:property|name)=["']${propertyOrName}["'][^>]+content=["']([^"']+)["'][^>]*>`,
    "i"
  );
  const m = html.match(propRe);
  return m?.[1]?.trim() || null;
}

function absolutizeUrl(baseUrl, maybeRelative) {
  try {
    return new URL(maybeRelative, baseUrl).toString();
  } catch {
    return null;
  }
}

function pickFirstLargeImage(html, baseUrl) {
  // fallback: first <img src="..."> that isn't a tiny icon
  const imgRe = /<img[^>]+src=["']([^"']+)["'][^>]*>/gi;
  let m;
  while ((m = imgRe.exec(html))) {
    const src = m[1];
    if (!src) continue;
    if (src.includes("svg")) continue;
    if (src.toLowerCase().includes("icon")) continue;
    return absolutizeUrl(baseUrl, src);
  }
  return null;
}

async function main() {
  const products = JSON.parse(fs.readFileSync(INPUT_FILE, "utf8"));

  for (const p of products) {
    try {
      const html = await fetchText(p.url);

      const ogImage = pickMeta(html, "og:image");
      const twitterImage = pickMeta(html, "twitter:image");
      const bestImage =
        absolutizeUrl(p.url, ogImage) ||
        absolutizeUrl(p.url, twitterImage) ||
        pickFirstLargeImage(html, p.url);

      // Optional: title/description enrichment
      const ogTitle = pickMeta(html, "og:title");
      const metaDesc = pickMeta(html, "description");

      p.imageUrl = p.imageUrl || bestImage || null;
      p.name = p.name || ogTitle || p.name;
      p.shortDescription = p.shortDescription || metaDesc || p.shortDescription;

      console.log(`✓ ${p.id} -> ${p.imageUrl ?? "NO IMAGE FOUND"}`);
    } catch (e) {
      console.warn(`⚠ ${p.id} (${p.url}) -> ${e.message}`);
      p.imageUrl = p.imageUrl ?? null;
    }
  }

  fs.writeFileSync(OUTPUT_FILE, JSON.stringify(products, null, 2));
  console.log(`\nDone. Wrote: ${OUTPUT_FILE}`);
}

main();

How to use it

Save your catalog array as JSON (example: copy products into a file and JSON.stringify it once), or manually create productCatalog.seed.json as a JSON array.

Run: node scrapeProductImages.mjs

Import productCatalog.withImages.json in your app.
